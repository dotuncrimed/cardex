rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ─── HELPERS ──────────────────────────────────────────────

    function signedIn() {
      return request.auth != null;
    }

    function isSameUsername(username) {
      return signedIn() &&
        request.auth.token.email == username + '@chinesepoker.app';
    }

    // ─── USERS ────────────────────────────────────────────────

    match /users/{username} {
      // User can read their own doc
      // Signed-in users can also read others (needed for cash display)
      allow read: if signedIn();

      // Only the user themselves can create their doc
      allow create: if isSameUsername(username);

      // MVP: Allow any signed-in user to update
      // This is REQUIRED for multiplayer settlement
      // In production, move settlement to Cloud Functions
      allow update: if signedIn();

      // No one can delete users
      allow delete: if false;

      // ─── TRANSACTIONS SUBCOLLECTION ──────────────────────

      match /transactions/{transactionId} {
        allow read: if signedIn();
        allow create: if signedIn();
        allow update: if false;
        allow delete: if false;
      }
    }

    // ─── ROOMS ────────────────────────────────────────────────

    match /rooms/{roomId} {
      // Any signed-in user can read rooms (lobby, spectate)
      allow read: if signedIn();

      // Any signed-in user can create a room
      allow create: if signedIn();

      // Any signed-in user can update rooms
      // (join, submit, ready, start, finish)
      allow update: if signedIn();

      // Any signed-in user can delete rooms
      // (host leaving, stale room sweep)
      allow delete: if signedIn();

      // ─── HANDS SUBCOLLECTION ─────────────────────────────

      match /hands/{uid} {
        allow read: if signedIn();
        allow create: if signedIn();
        allow update: if signedIn();
        allow delete: if signedIn();
      }
    }

    // ─── ADMIN LOGS ───────────────────────────────────────────

    match /adminLogs/{logId} {
      allow read: if signedIn();
      allow create: if signedIn();
      allow update: if false;
      allow delete: if false;
    }
  }
}
